cmake_minimum_required(VERSION 3.20)
project(dfttestvk VERSION 0.1.1 LANGUAGES CXX)
set(VAPOURSYNTH_INCLUDE_DIR "" CACHE PATH "VapourSynth R81 headers")
set(VULKAN_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/vulkan-headers/include" CACHE PATH "Vulkan-Headers include directory")
if(NOT VAPOURSYNTH_INCLUDE_DIR)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import vapoursynth as vs; print(vs.get_include())"
        RESULT_VARIABLE result OUTPUT_VARIABLE directory OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(NOT result EQUAL 0)
        message(FATAL_ERROR "Select Python with VapourSynth R81 or set VAPOURSYNTH_INCLUDE_DIR")
    endif()
    set(VAPOURSYNTH_INCLUDE_DIR "${directory}")
endif()
foreach(header VapourSynth4.h VSVulkan4.h)
    if(NOT EXISTS "${VAPOURSYNTH_INCLUDE_DIR}/${header}")
        message(FATAL_ERROR "Missing ${header}; use the matching R81 SDK")
    endif()
endforeach()
if(NOT EXISTS "${VULKAN_HEADERS}/vulkan/vulkan_core.h")
    message(FATAL_ERROR "Run tools/fetch_dependencies.py or set VULKAN_HEADERS")
endif()
add_library(dfttestvk MODULE src/dfttestvk.cpp)
target_compile_features(dfttestvk PRIVATE cxx_std_20)
target_compile_definitions(dfttestvk PRIVATE VS_USE_API_43)
target_include_directories(dfttestvk PRIVATE "${VAPOURSYNTH_INCLUDE_DIR}" "${VULKAN_HEADERS}")
set_target_properties(dfttestvk PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
    target_compile_options(dfttestvk PRIVATE /O2 /fp:precise)
else()
    target_compile_options(dfttestvk PRIVATE -O3)
    if(MINGW)
        target_link_options(dfttestvk PRIVATE -static -Wl,--no-insert-timestamp)
    endif()
endif()
install(TARGETS dfttestvk LIBRARY DESTINATION lib/vapoursynth RUNTIME DESTINATION lib/vapoursynth)
